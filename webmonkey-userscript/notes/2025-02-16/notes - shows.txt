brightcove API:
===============

usage:
======

policy_key=''
account_id=''
reference_id=''
referer_url=''

curl --insecure \
  -H "Referer: ${referer_url}" \
  -H "BCOV-POLICY: ${policy_key}" \
  "https://edge.api.brightcove.com/playback/v1/accounts/${account_id}/videos/ref:${reference_id}"

alternatively:
==============

curl --insecure \
  -H "Referer: ${referer_url}" \
  -H "Accept: application/json;pk=${policy_key}" \
  "https://edge.api.brightcove.com/playback/v1/accounts/${account_id}/videos/ref:${reference_id}"

  JSON.parse(
    decodeURIComponent(
      document.querySelector('meta[name="mw-3now/config/environment"][content]').getAttribute('content')
    )
  )
  .videoPlayers.map(player => {
    if (player && player.assets && Array.isArray(player.assets.scripts) && player.assets.scripts.length) {
      for (var i=0; i < player.assets.scripts.length; i++) {
        if (player.assets.scripts[i].indexOf('players.brightcove.net') >= 0)
          return player.assets.scripts[i]
      }
    }
    return null
  })
  .filter(url => !!url)
  .join("\n")
)
